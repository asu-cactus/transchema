# Separate columns by type for filling missing values and casting
string_cols = ['name', 'Gender', 'Eye color', 'Race', 'Hair color', 'Publisher', 'Skin color', 'Alignment', 'hero_names']
int_cols = ['Height']
float_cols = ['Weight']
# All other columns except above are boolean columns
bool_cols = [col for col in target_columns if col not in string_cols + int_cols + float_cols]

# Some boolean columns may not be present in df_merged if source1 had no data for some heroes
# Ensure all boolean columns exist in df_merged, add missing with default False
for col in bool_cols:
    if col not in df_merged.columns:
        df_merged[col] = False

# Fill missing values
# For string columns: fill with empty string
for col in string_cols:
    if col in df_merged.columns:
        df_merged[col] = df_merged[col].fillna('')
    else:
        # If column missing, create with empty string
        df_merged[col] = ''

# For int columns: convert to numeric, coerce errors, fill NaN with 0, then convert to int
for col in int_cols:
    if col in df_merged.columns:
        df_merged[col] = pd.to_numeric(df_merged[col], errors='coerce').fillna(0).astype(int)
    else:
        df_merged[col] = 0

# For float columns: convert to numeric, coerce errors, fill NaN with 0.0, keep float dtype
for col in float_cols:
    if col in df_merged.columns:
        df_merged[col] = pd.to_numeric(df_merged[col], errors='coerce').fillna(0.0).astype(float)
    else:
        df_merged[col] = 0.0

# For boolean columns: fill NaN with False and convert to bool
for col in bool_cols:
    df_merged[col] = df_merged[col].fillna(False)
    # Sometimes boolean columns may be read as object or int, convert explicitly
    df_merged[col] = df_merged[col].astype(bool)

# The join created both 'name' and 'hero_names' columns.
# According to target schema, both columns must be present.
# They may differ if outer join introduced unmatched rows.
# For rows where one is empty string, fill with the other if available.
def fill_name_hero_names(row):
    if row['name'] == '' and row['hero_names'] != '':
        return row['hero_names']
    return row['name']

def fill_hero_names_name(row):
    if row['hero_names'] == '' and row['name'] != '':
        return row['name']
    return row['hero_names']

df_merged['name'] = df_merged.apply(fill_name_hero_names, axis=1)
df_merged['hero_names'] = df_merged.apply(fill_hero_names_name, axis=1)

# Reorder columns exactly as per target schema
df_final = df_merged[target_columns]
